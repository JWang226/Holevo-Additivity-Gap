∀ {M : Nat},
  LT.lt.{0} 0 M →
    ∀ (a : Fin M → Real),
      (∀ (m : Fin M), LE.le.{0} 0 (a m)) →
        LE.le.{0} (HDiv.hDiv.{0, 0, 0} (∑ m, Real.sqrt (a m)) ↑M)
